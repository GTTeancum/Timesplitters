# Prepare Xbox Files: the Xbox button glyphs (xbox_glyphs/*.qpm, the Xbox
# build's src/xbox/assets/buttons/qpm) embedded as bytes in xbox_files.cpp.
file(GLOB TS_XBOX_GLYPHS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/xbox_glyphs/*.qpm")
set(ts_glyph_inc "// Generated by project/game/CMakeLists.txt from project/game/xbox_glyphs/*.qpm.\n")
set(ts_glyph_table "")
foreach(qpm IN LISTS TS_XBOX_GLYPHS)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${qpm}")
    get_filename_component(qpm_name "${qpm}" NAME)
    get_filename_component(qpm_stem "${qpm}" NAME_WE)
    file(READ "${qpm}" qpm_hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," qpm_bytes "${qpm_hex}")
    string(APPEND ts_glyph_inc "static const unsigned char kGlyph_${qpm_stem}[] = {${qpm_bytes}};\n")
    string(APPEND ts_glyph_table "    {\"textures/misc/${qpm_name}\", kGlyph_${qpm_stem}, sizeof(kGlyph_${qpm_stem})},\n")
endforeach()
string(APPEND ts_glyph_inc "static const GlyphFile kGlyphFiles[] = {\n${ts_glyph_table}};\n")
set(ts_glyph_out "${CMAKE_CURRENT_BINARY_DIR}/xbox_glyphs/xbox_glyphs.inc")
set(ts_glyph_old "")
if(EXISTS "${ts_glyph_out}")
    file(READ "${ts_glyph_out}" ts_glyph_old)
endif()
if(NOT ts_glyph_old STREQUAL ts_glyph_inc)
    file(WRITE "${ts_glyph_out}" "${ts_glyph_inc}")
endif()
set_property(SOURCE xbox_files.cpp APPEND PROPERTY INCLUDE_DIRECTORIES "${CMAKE_CURRENT_BINARY_DIR}/xbox_glyphs")
set_property(SOURCE xbox_files.cpp APPEND PROPERTY OBJECT_DEPENDS "${ts_glyph_out}")

# Push B/C (src/xbox/PUSH-BC-PLAN.md W0-I; frozen afterwards): the Xbox
# natives, the game-side registries (hooks, knobs, list tokens, records,
# pacing) and the idiomatic modules under project/game/native, built into
# the PC game for the host-side tests and B-LOGIC's whole-run differential.
# Covers every project/game/ts_*.cpp (ts_overrides.cpp is in the game
# already) and project/game/native/*.cpp, present or still to come
# (ts_inplace.cpp, ts_record_*.cpp, ...): a track's new file of either name
# joins the PC build without touching this file. A file of another name (a
# tool with its own main, like frame_bench.cpp or tick_replay.cpp) is its
# own target through an integrator amendment, or stays Xbox-only.
# Off (default): the PC game is what it was.
option(TS_NATIVE_HOST "Build the Xbox natives and their harness into the PC game" OFF)
if(TS_NATIVE_HOST)
    file(GLOB TS_NATIVE_HOST_SRCS CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/ts_*.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/native/*.cpp")
    list(REMOVE_ITEM TS_NATIVE_HOST_SRCS "${CMAKE_CURRENT_SOURCE_DIR}/ts_overrides.cpp")
    target_sources(timesplitters PRIVATE ${TS_NATIVE_HOST_SRCS})
    target_compile_definitions(timesplitters PRIVATE TS_NATIVE_HOST=1)
    target_include_directories(timesplitters PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/native")
endif()
if(MSVC)
    set_target_properties(timesplitters PROPERTIES UNITY_BUILD OFF)
    # /d2OptimizeHugeFunctions: MSVC otherwise silently skips optimizing the
    # largest recompiled functions (bulletTick, playerTick, animMake, ...).
    target_compile_options(timesplitters PRIVATE $<$<CONFIG:Debug>:/Od> /Zi /arch:AVX2 /bigobj
        $<$<NOT:$<CONFIG:Debug>>:/d2OptimizeHugeFunctions>)
    target_link_options(timesplitters PRIVATE /STACK:8388608)
else()
    set_target_properties(timesplitters PROPERTIES UNITY_BUILD ON UNITY_BUILD_BATCH_SIZE 32)
    target_compile_options(timesplitters PRIVATE -O1 -g1 -msse4.1 -mavx2)
endif()
